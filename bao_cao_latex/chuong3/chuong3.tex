\chapter[\MakeUppercase{Thiết kế luồng truy xuất tri thức}]{Thiết kế luồng truy xuất tri thức}
\label{ch:chuong3}

Chatbot truy xuất tri thức từ hai nguồn. Nguồn thứ nhất là giáo trình đã số hóa, truy xuất bằng tìm kiếm ngữ nghĩa, tìm kiếm từ khóa và luồng hỗn hợp (mục~\ref{sec:truy-xuat-tai-lieu}). Nguồn thứ hai là đồ thị tri thức (mục~\ref{sec:truy-xuat-do-thi}). Đoạn giáo trình và quan hệ trích từ giáo trình (có câu trích và số trang) là bằng chứng. Quan hệ của đồ thị nền PrimeKG và DermO chỉ là gợi ý: chúng cung cấp bệnh ứng viên, triệu chứng, thuốc, chống chỉ định và chuẩn hóa thuật ngữ để nhận diện thực thể.

\section{Truy xuất tài liệu}
\label{sec:truy-xuat-tai-lieu}

\subsection{Tổng quan}

Hệ thống có hai hướng tìm kiếm trên các đoạn giáo trình (chunk). Mỗi hướng bỏ sót những trường hợp mà hướng kia bắt được.

Ví dụ: câu hỏi ``da tôi đỏ và ngứa, nổi thành mảng'' và đoạn chứa ``erythematous pruritic plaques'' không có từ chung. BM25 cho điểm $0$, tìm kiếm ngữ nghĩa vẫn khớp. Ngược lại, câu hỏi về một tên thuốc cụ thể được BM25 khớp đúng chữ, trong khi vector có thể gộp thuốc đó với các thuốc cùng nhóm.

Luồng hỗn hợp chạy cả hai hướng, gộp kết quả và xếp hạng lại. Mục tiêu của tầng thứ nhất là không bỏ sót đoạn liên quan (recall). Mục tiêu của tầng xếp hạng lại là đưa đoạn đúng lên đầu (precision).

\subsection{Luồng tìm kiếm ngữ nghĩa}

\subsubsection{Nguyên lý}

Tìm kiếm ngữ nghĩa so khớp theo nghĩa của câu thay vì theo ký tự. Mô hình embedding đổi mỗi đoạn văn bản thành một vector thực $d$ chiều. Bài toán tra cứu trở thành bài toán hình học: tìm các vector đoạn gần vector câu hỏi nhất.

Hệ thống dùng mô hình \texttt{qwen/qwen3-embedding-8b} (8 tỷ tham số) qua OpenRouter, đầu ra $d=4096$ chiều. Mô hình có hai đặc tính:
\begin{itemize}
  \item \textbf{Instruction-tuned:} câu hỏi được thêm tiền tố chỉ dẫn; đoạn trong kho tri thức được mã hóa ở dạng nguyên bản.
  \item \textbf{Đa ngôn ngữ:} câu hỏi tiếng Việt và đoạn tiếng Anh nằm chung một không gian vector.
\end{itemize}

\subsubsection{Cơ chế hoạt động}

Gọi $f$ là hàm embedding. Quy trình có hai giai đoạn:
\begin{itemize}
  \item \textbf{Nạp dữ liệu (indexing):} mỗi đoạn được mã hóa một lần, $d=f(\text{đoạn})$, rồi lưu vào Qdrant.
  \item \textbf{Truy vấn (querying):} câu hỏi được mã hóa một lần, $q=f(\text{câu hỏi})$. Kiến trúc Bi-Encoder mã hóa câu hỏi và đoạn độc lập, nên vector đoạn được tính trước và độ trễ lúc truy vấn thấp.
\end{itemize}

Độ liên quan giữa $q$ và $d$ là độ tương đồng cosine:
\begin{equation}
  \mathrm{sim}(q,d)=\frac{q\cdot d}{\Vert q\Vert \Vert d\Vert }
\end{equation}

Trong đó, $\mathrm{sim}(q,d)\in[-1,1]$. Phép đo chỉ phụ thuộc hướng vector nên không bị lệch bởi độ dài đoạn.

Hệ thống xếp các đoạn theo $\mathrm{sim}$ giảm dần, lấy $k$ đoạn đầu, đọc nội dung và siêu dữ liệu (sách, mục, trang) theo ID vector rồi trả về.

\subsection{Luồng tìm kiếm từ khóa}

\subsubsection{Nguyên lý}

Tìm kiếm từ khóa khớp chính xác chuỗi ký tự. Luồng này bù cho điểm yếu của vector: tên thuốc, hoạt chất, mã bệnh và liều lượng thường bị làm mờ trong embedding.

\textbf{TF-IDF} là nền tảng của phương pháp. Điểm của tài liệu $D$ với câu hỏi $Q$ là tổng, trên các từ $t\in Q$, của tích tần suất từ trong tài liệu ($\mathrm{tf}$) và độ hiếm của từ trong kho ($\mathrm{idf}$):
\begin{equation}
  \mathrm{score}_{\mathrm{TF\text{-}IDF}}(D,Q)=\sum _{t\in Q} \mathrm{tf}(t,D)\cdot \mathrm{idf}(t)
\end{equation}

TF-IDF có hai hạn chế:
\begin{itemize}
  \item \textbf{Không bão hòa:} điểm tăng tuyến tính theo tần suất từ, dù từ lặp lại nhiều lần không làm đoạn liên quan hơn tương ứng.
  \item \textbf{Không chuẩn hóa độ dài:} đoạn dài chứa nhiều từ hơn nên được điểm cao hơn.
\end{itemize}

\subsubsection{Thuật toán BM25}

BM25 (Best Matching 25) thêm hai tham số để sửa hai hạn chế trên:
\begin{equation}
  \mathrm{score}(D,Q)=\sum_{t\in Q}\mathrm{IDF}(t)\cdot\frac{\mathrm{tf}\cdot(k_1+1)}{\mathrm{tf}+k_1\cdot\left(1-b+b\cdot\frac{|D|}{\mathrm{avgdl}}\right)}
\end{equation}
\begin{equation}
  \mathrm{IDF}(t)=\ln\left( 1+\frac{N-n(t)+0.5}{n(t)+0.5}\right)
\end{equation}

Trong đó:
\begin{itemize}
  \item $\mathrm{tf}(t,D)$: số lần từ $t$ xuất hiện trong $D$; $n(t)$: số đoạn chứa $t$; $N$: tổng số đoạn.
  \item $k_1=1.2$: giới hạn mức tăng điểm theo tần suất từ (độ bão hòa).
  \item $b=0.75$: mức chuẩn hóa theo tỷ số giữa độ dài đoạn $|D|$ và độ dài trung bình $\mathrm{avgdl}$.
\end{itemize}

Qdrant lưu phần $\mathrm{tf}$ của mỗi đoạn dưới dạng vector thưa (sparse) và chấm điểm khi truy vấn. Cấu trúc này hoạt động như chỉ mục đảo: mỗi từ trỏ tới các đoạn chứa nó, nên truy vấn không phải duyệt toàn bộ kho.

\subsubsection{Cơ chế hoạt động}

\begin{itemize}
  \item \textbf{Mở rộng câu hỏi Việt -- Anh:} LLM trích thực thể y khoa từ câu hỏi và thêm tên tiếng Anh, tiếng Việt vào câu hỏi BM25, để khớp với sách viết khác ngôn ngữ câu hỏi. Khi LLM lỗi, BM25 dùng câu hỏi gốc và hệ thống ghi chú lại.
  \item \textbf{Tiền xử lý tiếng Việt:} tách từ ghép bằng underthesea, loại stopword nhưng giữ từ phủ định và đơn vị liều. Thêm token bỏ dấu với trọng số thấp ($0.3$) để câu hỏi không dấu vẫn khớp.
  \item \textbf{Cách dùng:} chạy độc lập khi cần tra tên chính xác, và là một nhánh của luồng hỗn hợp.
\end{itemize}

\subsection{Luồng hỗn hợp}

\subsubsection{Quy trình}

Luồng hỗn hợp kết hợp hai luồng ở trên theo bốn bước:
\begin{enumerate}
  \item Trích thực thể từ câu hỏi và mở rộng câu hỏi cho nhánh BM25.
  \item Chạy song song nhánh ngữ nghĩa và nhánh BM25, mỗi nhánh lấy 20 đoạn.
  \item Gộp hai danh sách bằng Reciprocal Rank Fusion (RRF), khử trùng theo ID đoạn, giữ 20 đoạn đầu.
  \item Xếp hạng lại bằng Cross-Encoder, trả về $k$ đoạn đầu kèm sách, mục, số trang.
\end{enumerate}

Khi một nhánh lỗi, hệ thống trả kết quả của nhánh còn lại và ghi chú để agent báo rõ nguồn chưa tra được.

\subsubsection{Gộp kết quả bằng RRF}

RRF gộp theo thứ hạng thay vì điểm thô, vì điểm cosine và điểm BM25 có thang khác nhau:
\begin{equation}
  \mathrm{RRF}(d)=\sum_{i\in \mathrm{lists}} \frac{1}{k+\mathrm{rank}_{i}(d)},\quad k=60
\end{equation}

Đoạn xuất hiện ở cả hai danh sách được cộng hai số hạng nên lên cao. Hằng số $k=60$ làm giảm chênh lệch giữa các vị trí đầu, nên không nhánh nào chi phối hoàn toàn. Hệ thống không dùng phương pháp chuẩn hóa điểm có trọng số $\alpha$ vì phương pháp này nhạy với giá trị ngoại lai trong từng truy vấn và cần chỉnh $\alpha$ bằng thực nghiệm, trong khi chưa có tập đánh giá đủ lớn.

\subsubsection{Xếp hạng lại bằng Cross-Encoder}

Mô hình \texttt{qwen/qwen3-reranker-8b} nhận từng cặp $(\text{câu hỏi},\text{đoạn})$ và cho điểm từ $0$ đến $1$. Khác Bi-Encoder, Cross-Encoder đọc câu hỏi và đoạn cùng lúc nên đánh giá được tương tác giữa hai chuỗi, nhưng chậm hơn. Vì vậy mô hình chỉ chạy trên 20 đoạn sau RRF. Khi reranker không khả dụng, hệ thống giữ thứ tự RRF và ghi chú \texttt{ranked\_by = rrf}.

\subsection{Đánh giá luồng hỗn hợp}

\subsubsection{Bộ dữ liệu}

Bộ dữ liệu gồm 378 chunk từ sách \textit{Hướng dẫn chẩn đoán điều trị Da liễu} (Bộ Y tế) lưu trong Qdrant. Biểu diễn vector dùng \texttt{qwen/qwen3-embedding-8b}; xếp hạng lại dùng \texttt{qwen/qwen3-reranker-8b}.

Quy trình sinh bộ test:
\begin{enumerate}
  \item Chọn 120 chunk hợp lệ, loại chunk ngắn hơn 300 ký tự, mục lục và phụ lục.
  \item Dùng RAGAS dựng đồ thị tri thức và dùng LLM sinh 42 câu hỏi theo vai bệnh nhân (đơn bước và đa bước), kèm nhãn \texttt{reference\_contexts}.
  \item Kết quả: 24 câu hỏi (10 đơn bước, 14 đa bước) và 36 nhãn chunk.
\end{enumerate}

Hạn chế của bộ test:
\begin{itemize}
  \item Cỡ mẫu nhỏ ($n=24$), khoảng tin cậy rộng.
  \item Nhãn cố định theo RAGAS nên hit@k và recall@k là cận dưới.
  \item Câu hỏi sinh từ chính chunk nên văn phong gần văn bản gốc, thiên lợi cho BM25.
  \item Chưa được kiểm tra thủ công toàn bộ.
  \item Chưa đánh giá được truy vấn xuyên ngôn ngữ vì tài liệu chỉ có một ngôn ngữ.
\end{itemize}

\section{Truy xuất từ đồ thị tri thức}
\label{sec:truy-xuat-do-thi}

\subsection{Tổng quan}

Đồ thị tri thức lưu quan hệ có cấu trúc giữa bệnh, triệu chứng và thuốc. Tìm kiếm trên văn bản không trả lời tốt các câu hỏi dạng quan hệ như ``bệnh nào có triệu chứng này'' hay ``thuốc này chống chỉ định với bệnh nào''. Đồ thị trả lời các câu hỏi đó bằng truy vấn trực tiếp.

Đồ thị gồm hai thành phần nạp vào Neo4j:
\begin{itemize}
  \item \textbf{PrimeKG:} đồ thị tri thức y sinh có sẵn. Hệ thống giữ 5 loại nút (bệnh, thuốc, hiệu ứng/phenotype, gen/protein, phơi nhiễm) và 11 loại quan hệ như bệnh -- triệu chứng, bệnh -- thuốc, thuốc -- thuốc.
  \item \textbf{DermO:} ontology da liễu. Mỗi thuật ngữ có định nghĩa, từ đồng nghĩa và thuật ngữ cha.
\end{itemize}

Ngoài hai thành phần trên, pipeline số hóa giáo trình trích thêm các cạnh quan hệ từ sách và ghi vào cùng đồ thị, đánh dấu \texttt{source = DOC}.

Agent có hai công cụ truy vấn đồ thị, với hai mức tin cậy khác nhau:
\begin{itemize}
  \item \textbf{\texttt{knowledge\_graph\_query}} truy vấn PrimeKG và DermO. Kết quả mang nhãn \texttt{source = kg} và là \textbf{gợi ý}, không phải bằng chứng lâm sàng. Quan hệ trong đồ thị cho biết hai thực thể liên quan, không cho biết người bệnh mắc bệnh đó. DermO chỉ dùng để nhận diện thuật ngữ. Agent dùng công cụ này để chọn bệnh ứng viên, câu hỏi phân biệt và tra nhanh quan hệ thuốc -- bệnh, rồi đối chiếu bằng đoạn giáo trình trước khi kết luận.
  \item \textbf{\texttt{query\_document\_graph}} truy vấn các cạnh \texttt{DOC} trích từ giáo trình. Mỗi cạnh có câu trích gốc và số trang, nên kết quả được xếp vào \textbf{bằng chứng}. Cạnh không có câu trích bị bỏ, vì không có gì để trích dẫn.
\end{itemize}

\subsection{Nguồn của đồ thị}

\subsubsection{Đồ thị nền}

PrimeKG và DermO được tải về, làm sạch và nạp bằng các lệnh \texttt{make kg-download}, \texttt{kg-build}, \texttt{kg-load}. Bước làm sạch bỏ cạnh trùng và dòng lỗi. Lệnh \texttt{kg-verify} so số nút và số cạnh trong Neo4j với số liệu tổng hợp lúc làm sạch.

Hai đồ thị không có cạnh nối trực tiếp. Cầu nối duy nhất là so khớp tên: DermO chuẩn hóa tên, PrimeKG cung cấp quan hệ.

\subsubsection{Quan hệ trích từ giáo trình}

Pipeline số hóa có hai bước dành cho đồ thị:
\begin{enumerate}
  \item \textbf{Bước \texttt{graph}:} LLM đọc từng mục sách và điền vào các ô quan hệ theo lược đồ do quản trị viên định nghĩa. Tên thực thể trong sách được khớp với nút của PrimeKG hoặc DermO qua bảng từ vựng. Kết quả là các bộ ba (thực thể, quan hệ, thực thể). Người duyệt xem và sửa các bộ ba này.
  \item \textbf{Bước \texttt{graph\_index}:} chỉ nạp các bộ ba hợp lệ vào Neo4j, xóa cạnh cũ của cùng tài liệu trước khi ghi, và kiểm tra số cạnh đã lưu bằng số cạnh dự kiến.
\end{enumerate}

Mỗi cạnh \texttt{DOC} lưu câu trích dẫn (\texttt{evidence}) và số trang. Nhờ đó người dùng truy ngược được nguồn của quan hệ.

Cấu hình \texttt{KG\_EDGE\_SOURCES} (mặc định \texttt{["KG"]}) quy định \texttt{knowledge\_graph\_query} dùng nguồn cạnh nào. Công cụ \texttt{query\_document\_graph} không phụ thuộc cấu hình này và chỉ đọc cạnh \texttt{DOC}.

\subsection{Giải quyết thực thể}

Câu hỏi của bệnh nhân dùng từ đời thường nên phải ánh xạ về nút trong đồ thị trước khi truy vấn. Quy trình gồm ba bước:
\begin{enumerate}
  \item \textbf{Trích thực thể:} LLM trích bệnh, triệu chứng, thuốc từ câu hỏi, dịch sang tiếng Anh y khoa và giữ tên tiếng Việt.
  \item \textbf{Chuẩn hóa bằng DermO:} tra thuật ngữ DermO khớp chính xác tên hoặc từ đồng nghĩa, ưu tiên khớp chính xác trước khớp một phần. Kết quả là tên chuẩn và các từ đồng nghĩa.
  \item \textbf{Tìm nút gốc (seed):} dùng tập tên trên khớp nút PrimeKG thuộc loại bệnh, phenotype hoặc thuốc. Nếu không có khớp chính xác, hệ thống khớp một phần và lấy tối đa 3 nút.
\end{enumerate}

Id của PrimeKG chỉ duy nhất theo loại nút (8.822 id dùng chung cho nhiều loại), nên mọi truy vấn khớp nút kèm nhãn loại.

\subsection{Truy vấn theo ý định}

Công cụ \texttt{knowledge\_graph\_query} nhận một ý định (intent) và danh sách thực thể. Mỗi ý định ứng với một câu Cypher cố định, chỉ duyệt đúng nhóm quan hệ cần dùng. Cách này cho kết quả gọn, và không để LLM tự sinh Cypher.

\begin{table}[H]
\centering
\small
\begin{tabular}{|p{4.2cm}|p{4.8cm}|p{3.9cm}|}
\hline
\textbf{Ý định} & \textbf{Quan hệ duyệt} & \textbf{Thực thể đầu vào} \\ \hline
\texttt{disease\_overview} & Triệu chứng, vị trí thường gặp, yếu tố nguy cơ, dấu hiệu cảnh báo & Bệnh \\ \hline
\texttt{symptoms\_of\_disease} & Triệu chứng có và không có & Bệnh \\ \hline

\texttt{contraindications} & Chống chỉ định & Bệnh \\ \hline
\texttt{drug\_profile} & Chỉ định, chống chỉ định, dùng ngoài chỉ định, tác dụng phụ, tương tác thuốc & Thuốc \\ \hline

Quy tắc chung của các truy vấn:
\begin{itemize}
  \item Cạnh duyệt không phân biệt chiều và lọc theo nguồn cạnh được phép.
  \item Mỗi quan hệ trả tối đa một số nút nhất định, sắp theo tên để kết quả ổn định giữa các lần gọi.
  \item Mỗi dữ kiện trả về dạng \{ý định, quan hệ, văn bản\}, kèm trang sách nếu cạnh có nguồn.
  \item Ý định cần loại thực thể nhất định. Nếu thực thể không khớp loại, công cụ trả ghi chú cho agent thay vì kết quả sai.
\end{itemize}

\subsection{Truy vấn đồ thị trích từ giáo trình}

Công cụ \texttt{query\_document\_graph} nhận tên một thực thể (tiếng Anh hoặc tiếng Việt) và một loại quan hệ tùy chọn, ví dụ \texttt{INDICATION}, \texttt{CONTRAINDICATION}. Quy trình:
\begin{enumerate}
  \item Tìm tối đa 5 nút (thực thể PrimeKG hoặc thuật ngữ DermO) có tên hoặc từ đồng nghĩa chứa chuỗi đầu vào, không phân biệt hoa thường.
  \item Lấy các cạnh \texttt{DOC} một bước từ các nút đó, theo cả hai chiều, lọc theo loại quan hệ nếu có, tối đa 50 cạnh.
  \item Trả về chủ thể, quan hệ, đối tượng, câu trích, số trang và mã tài liệu.
\end{enumerate}

Việc khớp theo chuỗi nên tên không có trong đồ thị cho kết quả rỗng. Khi đó công cụ trả ``không tìm thấy quan hệ'' và agent dùng nguồn khác.

\subsection{Gợi ý bệnh ứng viên và câu hỏi phân biệt}
\label{sec:ung-vien}

Hai ý định \texttt{diseases\_with\_symptom} và \texttt{differential} chấm điểm bệnh ứng viên:
\begin{itemize}
  \item \textbf{Liên kết 1 bước:} bệnh có trực tiếp triệu chứng được hỏi, bệnh liên quan, chỉ định thuốc.
  \item \textbf{Liên kết 2 bước:} bệnh có chung phenotype với bệnh đầu vào. Điểm của nhóm này có trần ($2.0$) để bệnh có nhiều triệu chứng toàn thân không lấn át tín hiệu trực tiếp.
  \item Kết quả loại chính bệnh đầu vào.
\end{itemize}

Khi agent đang cân nhắc từ hai bệnh trở lên, hệ thống lấy tập triệu chứng của từng bệnh và chọn phenotype phân biệt tốt nhất. Phenotype xuất hiện ở ít bệnh (bậc thấp trong đồ thị) phân biệt tốt hơn phenotype xuất hiện ở nhiều bệnh. Agent dùng các phenotype này để hỏi bệnh nhân câu tiếp theo.

Điểm của đồ thị chỉ phản ánh số liên kết, không phải xác suất chẩn đoán.

\subsection{Các nơi khác dùng đồ thị}

\begin{itemize}
  \item \textbf{Tra DermO:} công cụ \texttt{lookup\_dermo\_term} trả định nghĩa, từ đồng nghĩa và thuật ngữ cha của một thuật ngữ.
  \item \textbf{Mở rộng câu hỏi BM25:} bước trích thực thể ở luồng hỗn hợp dùng chung với đồ thị (mục~\ref{sec:truy-xuat-tai-lieu}).
\end{itemize}

\section{Kết luận}

Truy xuất từ giáo trình dùng hai hướng bổ sung nhau. Tìm kiếm ngữ nghĩa bắt các cách diễn đạt khác nhau và lệch ngôn ngữ. BM25 bắt chính xác tên thuốc và thuật ngữ. Luồng hỗn hợp gộp hai hướng bằng RRF rồi xếp hạng lại bằng Cross-Encoder. Trên bộ kiểm thử 24 câu, luồng hỗn hợp đạt hit@5 = 1.00, recall@10 = 0.94 và MRR@10 = 0.90, cao hơn cả hai luồng đơn. Bộ kiểm thử nhỏ và đơn ngữ nên các số liệu mang tính tham khảo.

Truy xuất từ đồ thị trả lời các câu hỏi quan hệ bằng truy vấn Cypher theo ý định. Hệ thống ánh xạ thực thể qua DermO rồi truy vấn PrimeKG, và gợi ý bệnh ứng viên cùng câu hỏi phân biệt. Kết quả từ PrimeKG và DermO chỉ là gợi ý và cần đối chiếu với giáo trình. Quan hệ trích từ giáo trình, có câu trích và số trang, cùng đoạn giáo trình là bằng chứng.
